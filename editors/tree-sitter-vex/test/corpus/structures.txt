struct Particle {
    vector position;
    float mass;
    float scaled(float factor) { return mass * factor; }
};
function float[] weights(int count;) {
    float result[];
    for (int i = 0; i < count; i++) {
        append(result, float(i));
    }
    return result;
}

    vector result = (vector)input;
    result = normalize(result) * 2 + {1, 0, 0};
    do { result.x += 1; } while (result.x < 0);
    while (length(result) > 100) { result *= .5; }
    return result.x > 0 ? result : -result;
